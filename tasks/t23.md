Context:
- src/script.c is the Lua embedding layer with zero targeted test coverage.
- Existing E2E tests (smoke/latency/reconnect/errors) exercise some script.*
  functions implicitly, but ONLY their default/no-op paths because no test
  passes a -s script file. The following are NOT covered at all:
    - script_response()         — requires response() hook in user Lua
    - script_want_response() → true path — same
    - script_has_done() → true path    — requires done() hook
    - script_summary(), script_errors(), script_done() — gated by has_done()
    - script_init() setup() sub-path   — requires user setup() function
    - script_init() init() sub-path    — requires user init() function
    - dynamic request() path            — script_is_static() → false
  These need dedicated E2E tests with -s Lua fixture scripts.
- LuaJIT + ASAN conflict: test_script binary must NOT be added to test-asan.

Test cases (all use: char *headers[] = {NULL}; as empty headers):

  test_script_create_url_fields
    L = script_create(NULL, "http://example.com:9090/api", headers)
    assert wrk.scheme == "http", wrk.host == "example.com",
           wrk.port == "9090", wrk.path == "/api"
    lua_close(L)

  test_script_create_no_file_succeeds
    L = script_create(NULL, "http://localhost/", headers)
    TEST_ASSERT_NOT_NULL(L); lua_close(L)

  test_script_request_default
    L = script_create(NULL, "http://localhost/", headers)
    char *buf = NULL; size_t len = 0;
    script_request(L, &buf, &len)
    assert buf != NULL, len > 0
    assert strstr(buf, "GET") != NULL
    assert strstr(buf, "HTTP/1.1") != NULL
    free(buf); lua_close(L)

  test_script_request_custom
    L = script_create; luaL_dostring(L, "function request() return 'PING' end")
    script_request(L, &buf, &len)
    assert memcmp(buf, "PING", 4) == 0
    free(buf); lua_close(L)

  test_script_is_static_default
    L = script_create(NULL, ...)
    TEST_ASSERT_TRUE(script_is_static(L))
    lua_close(L)

  test_script_is_static_dynamic
    luaL_dostring(L, "function request() return wrk.format() end")
    TEST_ASSERT_FALSE(script_is_static(L))
    lua_close(L)

  test_script_want_response_false
    L = script_create(NULL, ...) — no response() defined
    TEST_ASSERT_FALSE(script_want_response(L))

  test_script_want_response_true
    luaL_dostring(L, "function response(s,h,b) end")
    TEST_ASSERT_TRUE(script_want_response(L))

  test_script_has_done_false
    L = script_create(NULL, ...) — no done() defined
    TEST_ASSERT_FALSE(script_has_done(L))

  test_script_has_done_true
    luaL_dostring(L, "function done(s,l,r) end")
    TEST_ASSERT_TRUE(script_has_done(L))

  test_script_parse_url_fields
    struct http_parser_url parts = {};
    script_parse_url("http://host.com:8080/foo", &parts)
    assert parts.field_set has UF_SCHEMA, UF_HOST, UF_PORT, UF_PATH bits set

  test_buffer_append_and_reset
    buffer b = {0};
    buffer_append(&b, "hello", 5)
    assert (b.cursor - b.buffer) == 5
    buffer_reset(&b)
    assert b.cursor == b.buffer  (reset to start)
    free(b.buffer)

--- Fixture scripts ---

tests/e2e/scripts/hooks.lua
  Exercises: init(), setup(), response(), done(), summary(), errors()
  All hooks print sentinel markers via io.write() inside done():

    local n_init      = 0
    local n_setup     = 0
    local n_responses = 0

    function init(args)     n_init  = n_init  + 1 end
    function setup(thread)  n_setup = n_setup + 1 end
    function response(status, headers, body)
        n_responses = n_responses + 1
    end
    function done(summary, latency, requests)
        io.write("HOOK:init="      .. tostring(n_init  > 0) .. "\n")
        io.write("HOOK:setup="     .. tostring(n_setup > 0) .. "\n")
        io.write("HOOK:responses=" .. tostring(n_responses) .. "\n")
    end

tests/e2e/scripts/dynamic_request.lua
  Exercises: dynamic request() path (script_is_static → false)

    local paths = {"/", "/health", "/index"}
    local i = 0
    function request()
        i = (i % #paths) + 1
        return wrk.format("GET", paths[i])
    end
