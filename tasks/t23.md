Context:
- src/script.c is the Lua embedding layer (the only source file with zero test
  coverage). It exposes the wrk Lua API: URL field injection, request/response
  hooks, thread setup, stats access, URL parsing, and buffer utilities.
- script_create() loads wrk.lua via `require "wrk"` — the bytecode is linked
  from obj/bytecode.o. The test binary must link against both libluajit.a and
  obj/bytecode.o.
- LuaJIT + ASAN conflict (custom mmap allocator). test_script must NOT be
  added to the test-asan target. Add only to test-unit.

Public API under test (src/script.h):
  script_create()       — create Lua state, inject wrk.* URL fields
  script_request()      — call Lua request() / wrk.request()
  script_is_static()    — true if request is prebuilt (not a dynamic function)
  script_want_response()— true if response() hook is defined
  script_has_done()     — true if done() hook is defined
  script_verify_request()— validates raw HTTP request bytes
  script_parse_url()    — wraps http_parser_parse_url
  buffer_append()       — append bytes to a buffer struct
  buffer_reset()        — reset cursor/length of a buffer struct

Steps:

1. Create tests/unit/test_script.c with the following test cases:

  test_script_create_url_fields
    - char *headers[] = {NULL};
    - lua_State *L = script_create(NULL, "http://example.com:9090/api", headers);
    - lua_getglobal(L, "wrk"); lua_getfield(L, -1, "scheme");
    - TEST_ASSERT_EQUAL_STRING("http", lua_tostring(L, -1));
    - Similarly assert host == "example.com", port == "9090", path == "/api"
    - lua_close(L);

  test_script_create_no_file_succeeds
    - script_create(NULL, "http://localhost/", headers) must return non-NULL
    - lua_close(L);

  test_script_request_default
    - script_create(NULL, "http://localhost/", headers)
    - char *buf = NULL; size_t len = 0;
    - script_request(L, &buf, &len);
    - TEST_ASSERT_NOT_NULL(buf); TEST_ASSERT_TRUE(len > 0);
    - assert buf contains "GET" and "HTTP/1.1"
    - free(buf); lua_close(L);

  test_script_request_custom
    - script_create; then luaL_dostring(L, "function request() return 'CUSTOM' end");
    - script_request(L, &buf, &len);
    - assert buf == "CUSTOM"

  test_script_is_static_default
    - script_create with no file → script_is_static(L) must return true
    - (wrk.request is pre-built by wrk.init; no dynamic function)

  test_script_is_static_dynamic
    - luaL_dostring(L, "function request() return wrk.format() end");
    - script_is_static(L) must return false

  test_script_want_response_false
    - script_create with no file → script_want_response(L) == false

  test_script_want_response_true
    - luaL_dostring(L, "function response(s,h,b) end");
    - script_want_response(L) == true

  test_script_has_done_false
    - script_create with no file → script_has_done(L) == false

  test_script_has_done_true
    - luaL_dostring(L, "function done(s,r) end");
    - script_has_done(L) == true

  test_script_parse_url_fields
    - struct http_parser_url parts = {};
    - script_parse_url("http://host.com:8080/foo", &parts);
    - assert UF_SCHEMA, UF_HOST, UF_PORT, UF_PATH bits are set in parts.field_set

  test_buffer_append_and_reset
    - buffer b = {0}; buffer_append(&b, "hello", 5);
    - TEST_ASSERT_EQUAL_INT(5, b.cursor - b.buffer);
    - buffer_reset(&b); assert cursor == buffer (reset to start)
    - free(b.buffer);
